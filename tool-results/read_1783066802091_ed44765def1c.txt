    32→// ── Convert decimal hours to HH.MM display format ──
    33→function formatHours(decimal: number): string {
    34→  if (!decimal || decimal === 0) return '0:00';
    35→  const hours = Math.floor(decimal);
    36→  const minutes = Math.round((decimal - hours) * 60);
    37→  if (minutes >= 60) return `${hours + 1}:00`;
    38→  return `${hours}:${String(minutes).padStart(2, '0')}`;
    39→}
    40→
    41→// ── Format overtime in clear human-readable format ──
    42→function formatOT(decimal: number): string {
    43→  if (!decimal || decimal === 0) return '0m';
    44→  const totalMinutes = Math.round(decimal * 60);
    45→  if (totalMinutes < 60) return `${totalMinutes}m`;
    46→  const hours = Math.floor(totalMinutes / 60);
    47→  const minutes = totalMinutes % 60;
    48→  if (minutes === 0) return `${hours}h`;
    49→  return `${hours}h ${minutes}m`;
    50→}
    51→
    52→// ── Display a value that's already in HH:MM format ──
    53→function displayHHMM(value: number | undefined | null): string {
    54→  if (!value && value !== 0) return '0:00';
    55→  return formatHours(value);
    56→}

    58→// ── Get firm code from employee ID prefix ──
    59→function getFirmFromEmployeeId(employeeId: string): string {
    60→  const id = employeeId.toUpperCase();
    61→  if (id.startsWith('LAPL')) return 'LAPL';
    62→  if (id.startsWith('LRSL')) return 'LRSL';
    63→  if (id.startsWith('SI-') || id.startsWith('SI0')) return 'SI';
    64→  if (id.startsWith('SDF')) return 'SDF';
    65→  return ''; // fallback to existing department/firm
    66→}

   130→// ── Animated Counter ──
   131→function AnimatedCounter({ value, prefix = '', suffix = '', decimals = 0 }: {
   132→  value: number; prefix?: string; suffix?: string; decimals?: number;
   133→}) {
   134→  const [display, setDisplay] = useState(0);
   135→  useEffect(() => {
   136→    const duration = 1000;
   137→    const startTime = Date.now();
   138→    const step = () => {
   139→      const elapsed = Date.now() - startTime;
   140→      const progress = Math.min(elapsed / duration, 1);
   141→      const eased = 1 - Math.pow(1 - progress, 3);
   142→      setDisplay(value * eased);
   143→      if (progress < 1) requestAnimationFrame(step);
   144→    };
   145→    requestAnimationFrame(step);
   146→  }, [value]);
   147→
   148→  if (decimals > 0) {
   149→    return <span>{prefix}{display.toFixed(decimals)}{suffix}</span>;
   150→  }
   151→  return <span>{prefix}{Math.round(display).toLocaleString('en-IN')}{suffix}</span>;
   152→}
