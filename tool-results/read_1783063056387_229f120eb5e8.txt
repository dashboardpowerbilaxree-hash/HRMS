    48→    48→// ── Convert decimal hours to HH:MM display format ──
    49→    49→// e.g., 6.25 → "6:15", 9.5 → "9:30", 8.0 → "8:00"
    50→    50→// Uses colon (:) separator so it's clearly hours:minutes, NOT a decimal fraction
    51→    51→function formatHours(decimal: number): string {
    52→    52→  if (!decimal || decimal === 0) return '0:00';
    53→    53→  const hours = Math.floor(decimal);
    54→    54→  const minutes = Math.round((decimal - hours) * 60);
    55→    55→  if (minutes >= 60) return `${hours + 1}:00`;
    56→    56→  return `${hours}:${String(minutes).padStart(2, '0')}`;
    57→    57→}
    58→    58→
    59→    59→// ── Format overtime hours in clear human-readable format ──
    60→    60→// Under 1 hour: shows minutes only (e.g., 7m, 20m)
    61→    61→// 1+ hours: shows hours and minutes (e.g., 1h 30m, 2h 15m)
    62→    62→// This avoids the confusing decimal display like "0.07h" which looks like 4.2 min but means 7 min
    63→    63→function formatOT(decimal: number): string {
    64→    64→  if (!decimal || decimal === 0) return '0m';
    65→    65→  const totalMinutes = Math.round(decimal * 60);
    66→    66→  if (totalMinutes < 60) {
    67→    67→    return `${totalMinutes}m`;
    68→    68→  }
    69→    69→  const hours = Math.floor(totalMinutes / 60);
    70→    70→  const minutes = totalMinutes % 60;
    71→    71→  if (minutes === 0) return `${hours}h`;
    72→    72→  return `${hours}h ${minutes}m`;
    73→    73→}

    77→    77→  if (!value && value !== 0) return '0:00';
    78→    78→  return formatHours(value);

   170→   170→// ── Animated Counter ──
   171→   171→function AnimatedCounter({ value, prefix = '', suffix = '', decimals = 0 }: {
   172→   172→  value: number; prefix?: string; suffix?: string; decimals?: number;
   173→   173→}) {
   174→   174→  const [display, setDisplay] = useState(0);
   175→   175→  useEffect(() => {
   176→   176→    const duration = 1000;
   177→   177→    const startTime = Date.now();
   178→   178→    const step = () => {
   179→   179→      const elapsed = Date.now() - startTime;
   180→   180→      const progress = Math.min(elapsed / duration, 1);
   181→   181→      const eased = 1 - Math.pow(1 - progress, 3);
   182→   182→      setDisplay(value * eased);
   183→   183→      if (progress < 1) requestAnimationFrame(step);
   184→   184→    };
   185→   185→    requestAnimationFrame(step);
   186→   186→  }, [value]);
   187→   187→
   188→   188→  if (decimals > 0) {
   189→   189→    return <span>{prefix}{display.toFixed(decimals)}{suffix}</span>;
   190→   190→  }
   191→   191→  return <span>{prefix}{Math.round(display).toLocaleString()}{suffix}</span>;
   192→   192→}

   493→   493→  // ── Convert Excel serial date to YYYY-MM-DD ──
   494→   494→  const excelSerialToDate = (serial: number): string => {
   495→   495→    // Excel serial date: days since Jan 1, 1900 (with the 1900 leap year bug)
   496→   496→    const epoch = new Date(1899, 11, 30); // Dec 30, 1899 is day 0
   497→   497→    const date = new Date(epoch.getTime() + serial * 86400000);
   498→   498→    const y = date.getFullYear();
   499→   499→    const m = String(date.getMonth() + 1).padStart(2, '0');
   500→   500→    const d = String(date.getDate()).padStart(2, '0');
   501→   501→    return `${y}-${m}-${d}`;
   502→   502→  };
   503→   503→
   504→   504→  // ── Convert Excel decimal time to HH:MM ──
   505→   505→  const excelDecimalToTime = (value: any): string => {
   506→   506→    if (typeof value === 'string') {
   507→   507→      // Already a string - check if it looks like HH:MM
   508→   508→      const trimmed = value.trim();
   509→   509→      if (/^\d{1,2}:\d{2}(:\d{2})?$/.test(trimmed)) return trimmed.substring(0, 5);
   510→   510→      // Try parsing as number
   511→   511→      const num = parseFloat(trimmed);
   512→   512→      if (!isNaN(num) && num >= 0 && num < 1) {
   513→   513→        const totalMinutes = Math.round(num * 1440);
   514→   514→        const hours = Math.floor(totalMinutes / 60);
   515→   515→        const minutes = totalMinutes % 60;
   516→   516→        return `${String(hours).padStart(2, '0')}:${String(minutes).padStart(2, '0')}`;
   517→   517→      }
   518→   518→      return trimmed;
   519→   519→    }
   520→   520→    if (typeof value === 'number') {
   521→   521→      if (value >= 1) {
   522→   522→        // Could be a full datetime serial - extract time portion
   523→   523→        const timePart = value - Math.floor(value);
   524→   524→        const totalMinutes = Math.round(timePart * 1440);
   525→   525→        const hours = Math.floor(totalMinutes / 60);
   526→   526→        const minutes = totalMinutes % 60;
   527→   527→        return `${String(hours).padStart(2, '0')}:${String(minutes).padStart(2, '0')}`;
   528→   528→      }
   529→   529→      // Decimal time fraction (0 to <1)
   530→   530→      const totalMinutes = Math.round(value * 1440);
   531→   531→      const hours = Math.floor(totalMinutes / 60);
   532→   532→      const minutes = totalMinutes % 60;
   533→   533→      return `${String(hours).padStart(2, '0')}:${String(minutes).padStart(2, '0')}`;
   534→   534→    }
   535→   535→    return String(value || '');
   536→   536→  };
   537→   537→
   538→   538→  // ── Parse any date value to YYYY-MM-DD ──
   539→   539→  const parseDateValue = (value: any): string => {
   540→   540→    if (!value && value !== 0) return '';
   541→   541→    if (typeof value === 'number') {
   542→   542→      // Excel serial date number
   543→   543→      return excelSerialToDate(value);
   544→   544→    }
   545→   545→    const str = String(value).trim();
   546→   546→    if (!str) return '';
   547→   547→    // Already in YYYY-MM-DD format
   548→   548→    if (/^\d{4}-\d{2}-\d{2}$/.test(str)) return str;
   549→   549→    // Try DD/MM/YYYY or DD-MM-YYYY
   550→   550→    const dmyMatch = str.match(/^(\d{1,2})[/\-.](\d{1,2})[/\-.](\d{4})$/);
   551→   551→    if (dmyMatch) return `${dmyMatch[3]}-${dmyMatch[2].padStart(2, '0')}-${dmyMatch[1].padStart(2, '0')}`;
   552→   552→    // Try MM/DD/YYYY
   553→   553→    const mdyMatch = str.match(/^(\d{1,2})[/\-.](\d{1,2})[/\-.](\d{4})$/);
   554→   554→    if (mdyMatch) return `${mdyMatch[3]}-${mdyMatch[1].padStart(2, '0')}-${mdyMatch[2].padStart(2, '0')}`;
   555→   555→    // Fallback: try native Date parse
   556→   556→    try {
   557→   557→      const d = new Date(str);
   558→   558→      if (!isNaN(d.getTime())) {
   559→   559→        return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
   560→   560→      }
   561→   561→    } catch {}
   562→   562→    return str;
   563→   563→  };

   565→   565→  // ── Smart Employee ID matching ──
   566→   566→  const smartMatchEmployeeId = (rawId: string, empName: string): string => {
   567→   567→    if (!rawId && !empName) return '';
   568→   568→    let empId = String(rawId).trim();
   569→   569→
   570→   570→    // If we have an ID, try exact match first
   571→   571→    if (empId) {
   572→   572→      // Try exact match
   573→   573→      const exact = employees.find(e => e.employeeId === empId);
   574→   574→      if (exact) return exact.employeeId;
   575→   575→
   576→   576→      // Try with EMP- prefix (e.g., "41" → "EMP-041" or "EMP-41")
   577→   577→      const withPrefix = `EMP-${empId.padStart(3, '0')}`;
   578→   578→      const matchPad = employees.find(e => e.employeeId === withPrefix);
   579→   579→      if (matchPad) return matchPad.employeeId;
   580→   580→
   581→   581→      const withoutPad = `EMP-${empId}`;
   582→   582→      const matchNoPad = employees.find(e => e.employeeId === withoutPad);
   583→   583→      if (matchNoPad) return matchNoPad.employeeId;
   584→   584→    }
   585→   585→
   586→   586→    // Match by employee name
   587→   587→    if (empName) {
   588→   588→      const nameLower = empName.toLowerCase().trim();
   589→   589→      // Exact name match
   590→   590→      const exactName = employees.find(e => e.fullName.toLowerCase() === nameLower);
   591→   591→      if (exactName) return exactName.employeeId;
   592→   592→      // Partial name match
   593→   593→      const partialName = employees.find(e =>
   594→   594→        e.fullName.toLowerCase().includes(nameLower) || nameLower.includes(e.fullName.toLowerCase())
   595→   595→      );
   596→   596→      if (partialName) return partialName.employeeId;
   597→   597→    }
   598→   598→
   599→   599→    return empId; // Return whatever we have
   600→   600→  };
