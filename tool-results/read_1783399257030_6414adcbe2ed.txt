    44→function getFirmFromEmployeeId(employeeId: string): string {
    45→  const id = employeeId.toUpperCase();
    46→  if (id.startsWith('LAPL')) return 'LAPL';
    47→  if (id.startsWith('LRSL')) return 'LRSL';
    48→  if (id.startsWith('SI-') || id.startsWith('SI0')) return 'SI';
    49→  if (id.startsWith('SDF')) return 'SDF';
    50→  return '';
    51→}

    53→// HH:MM format matching Attendance Tracker's displayDecimalAsColon (e.g. 202.43 → "202:43")
    54→// This treats the decimal part as the minute display (2-digit zero-padded),
    55→// consistent with how the Attendance Tracker UI displays the same value.
    56→function displayDecimalAsColon(value: number | undefined | null): string {
    57→  if (value == null || isNaN(value as number)) return '0:00';
    58→  const [intPart, decPart] = Number(value).toFixed(2).split('.');
    59→  return `${intPart}:${decPart}`;
    60→}
    61→
    62→// HH:MM format for OT hours in summary (e.g. 1.5 → "1:30")
    63→function formatOTHHMM(decimal: number): string {
    64→  if (!decimal || decimal === 0) return '0:00';
    65→  const hours = Math.floor(decimal);
    66→  const minutes = Math.round((decimal - hours) * 60);
    67→  if (minutes >= 60) return `${hours + 1}:00`;
    68→  return `${hours}:${String(minutes).padStart(2, '0')}`;
    69→}

    76→    right:  { style, color: { rgb: color } },
    77→  };
    78→}

    85→// GET /api/payroll/summary-export?month=7&year=2026&firm=LAPL
    86→// ════════════════════════════════════════════════════════════
    87→export async function GET(request: NextRequest) {
    88→  try {
    89→    const { searchParams } = new URL(request.url);
    90→    const month = parseInt(searchParams.get('month') || String(new Date().getMonth() + 1));
    91→    const year  = parseInt(searchParams.get('year')  || String(new Date().getFullYear()));
    92→    const firm  = searchParams.get('firm') || '';
    93→
    94→    // ── Build query ──
    95→    const where: any = { month, year };
    96→    if (firm) {
    97→      const emps = await db.employee.findMany({
    98→        where: { firm },
    99→        select: { employeeId: true },
   100→      });
   101→      where.employeeId = { in: emps.map(e => e.employeeId) };
   102→    }
   103→
   104→    const payrolls = await db.payroll.findMany({
   105→      where,
   106→      include: {
   107→        employee: {
   108→          select: {
   109→            fullName: true,
   110→            employeeId: true,
   111→            firm: true,
   112→            department: true,
   113→            designation: true,
   114→            location: true,
   115→            salaryType: true,
   116→            hourlyRate: true,
   117→            shiftHours: true,
   118→            monthlySalary: true,
   119→            relievingDate: true,
   120→          },
   121→        },
   122→      },
   123→      orderBy: { employeeId: 'asc' },
   124→    });
   125→
   126→    const monthName = MONTHS[month - 1];
   127→    const generatedDate = new Date().toLocaleString('en-IN', {
   128→      day: '2-digit', month: 'numeric', year: 'numeric',
   129→      hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: true,
   130→    });
   131→
   132→    // ── Recompute live values for each payroll (so the export is always current) ──
   133→    const enriched = await Promise.all(payrolls.map(async (p) => {
   134→      const daysInMonth = new Date(p.year, p.month, 0).getDate();
   135→      const emp = p.employee;
   136→      const shiftHrs = emp?.shiftHours || 9;
   137→
   138→      const cutoffDay = getEffectiveCutoffDay(p.year, p.month, daysInMonth, emp?.relievingDate);
   139→
   140→      const hourlyRate = Math.round((p.monthlySalary / (daysInMonth * shiftHrs)) * 100) / 100;
   141→
   142→      // ── Attendance recalculation ──
   143→      const startDate = new Date(p.year, p.month - 1, 1);
   144→      const endDate = new Date(p.year, p.month, 1);
   145→
   146→      const [attendance, holidays] = await Promise.all([
   147→        db.attendance.findMany({
   148→          where: { employeeId: p.employeeId, date: { gte: startDate, lt: endDate } },
   149→        }),
   150→        db.holiday.findMany({ where: { date: { gte: startDate, lt: endDate } } }),
   151→      ]);
   152→
   153→      const elapsedHolidays = countHolidaysUpTo(holidays, cutoffDay);
   154→      const sundays = countSundaysUpTo(p.year, p.month, cutoffDay);
   155→      const totalWorkingDays = Math.max(0, cutoffDay - sundays - elapsedHolidays);
   156→
   157→      const effectiveAttendance = filterAttendanceUpTo(attendance, p.year, p.month, cutoffDay);
   158→
   159→      const rawPresentDays = effectiveAttendance.filter(a => ['present', 'late', 'early-out'].includes(a.status)).length;
   160→      const halfDays = effectiveAttendance.filter(a => a.status === 'half-day' || a.status === 'half_day').length;
   161→      const presentDays = rawPresentDays;
   162→
   163→      let totalBaseHours = 0;
   164→      let totalWorkMinutes = 0; // Sum of check-in/out durations (includes OT) — matches Attendance Tracker
   165→      for (const a of effectiveAttendance) {
   166→        // Sum work minutes from check-in/check-out (includes OT)
   167→        if (a.checkIn && a.checkOut) {
   168→          const [h1, m1] = a.checkIn.split(':').map(Number);
   169→          const [h2, m2] = a.checkOut.split(':').map(Number);
   170→          const workMin = Math.max(0, (h2 * 60 + m2) - (h1 * 60 + m1));
   171→          totalWorkMinutes += workMin;
   172→        }
   173→        // Base hours (excludes OT) — used internally for gross calc
   174→        if (['present', 'late', 'early-out', 'half-day', 'half_day'].includes(a.status)) {
   175→          const baseHrs = Math.max(0, (a.totalHours || 0) - (a.overtimeHours || 0));
   176→          totalBaseHours += baseHrs;
   177→        }
   178→      }
   179→      totalBaseHours = Math.round(totalBaseHours * 100) / 100;
   180→      // TRUE decimal hours (includes OT) — matches Attendance Tracker's totalWorkHours exactly
   181→      const totalWorkHoursDecimal = Math.round((totalWorkMinutes / 60) * 100) / 100;
   182→
   183→      const otHours = Math.round(
   184→        effectiveAttendance
   185→          .filter(a => ['present', 'late', 'half-day', 'half_day', 'early-out'].includes(a.status))
   186→          .reduce((sum, a) => sum + (a.overtimeHours || 0), 0) * 100
   187→      ) / 100;
   188→
   189→      // Leaves
   190→      const leaves = await db.leave.findMany({
   191→        where: { employeeId: p.employeeId, status: 'approved', startDate: { gte: startDate }, endDate: { lt: endDate } },
   192→      });
   193→
   194→      const holidayDateStrs = new Set(
   195→        holidays.map(h => {
   196→          const hd = new Date(h.date);
   197→          return `${hd.getFullYear()}-${String(hd.getMonth() + 1).padStart(2, '0')}-${String(hd.getDate()).padStart(2, '0')}`;
   198→        })
   199→      );
   200→      const presentDateStrs = new Set();
   201→      for (const a of effectiveAttendance) {
   202→        if (['present', 'late', 'early-out', 'half-day', 'half_day'].includes(a.status)) {
   203→          const ad = new Date(a.date);
   204→          presentDateStrs.add(`${ad.getFullYear()}-${String(ad.getMonth() + 1).padStart(2, '0')}-${String(ad.getDate()).padStart(2, '0')}`);
   205→        }
   206→      }
   207→
   208→      let effectivePaidLeaves = 0;
   209→      let effectiveUnpaidLeaves = 0;
   210→      const cutoffDate = new Date(p.year, p.month - 1, cutoffDay);
   211→      for (const leave of leaves) {
   212→        const isUnpaid = leave.type === 'unpaid' || leave.type === 'UL' || leave.type === 'LOP';
   213→        let d = new Date(leave.startDate);
   214→        const end = new Date(leave.endDate);
   215→        const effectiveEnd = end > cutoffDate ? cutoffDate : end;
   216→        while (d <= effectiveEnd) {
   217→          const dateStr = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
   218→          const isSunday = d.getDay() === 0;
   219→          const isHoliday = holidayDateStrs.has(dateStr);
   220→          if (!isSunday && !isHoliday && !presentDateStrs.has(dateStr)) {
   221→            if (isUnpaid) effectiveUnpaidLeaves++;
   222→            else effectivePaidLeaves++;
   223→          }
   224→          d.setDate(d.getDate() + 1);
   225→        }
   226→      }
   227→
   228→      const absentDays = Math.max(0, totalWorkingDays - presentDays - halfDays - effectivePaidLeaves - effectiveUnpaidLeaves);
   229→
   230→      const sundayHrs = sundays * shiftHrs;
   231→      const paidLeaveHrs = effectivePaidLeaves * shiftHrs;
   232→      const totalHrs = Math.round((totalBaseHours + sundayHrs + otHours + paidLeaveHrs) * 100) / 100;
   233→      const grossSalary = Math.round(hourlyRate * totalHrs * 100) / 100;
   234→      const otAmount = Math.round(otHours * hourlyRate * 100) / 100;
   235→      const netSalary = Math.round((grossSalary + (p.bonus || 0) + (p.incentive || 0) + (p.arrear || 0) - (p.totalDeductions || 0)) * 100) / 100;
   236→
   237→      const firmCode = getFirmFromEmployeeId(p.employeeId) || emp?.firm || emp?.department || '';
   238→      const firmDisplay = firmCode;
   239→
   240→      return {
   241→        employeeId: p.employeeId,
   242→        fullName: emp?.fullName || p.employeeId,
   243→        firm: firmDisplay,
   244→        monthlySalary: p.monthlySalary,
   245→        presentDays,
   246→        absentDays,
   247→        // workedHrs now matches Attendance Tracker's "Total Hrs Worked" exactly
   248→        // (true decimal, includes OT). Display uses displayDecimalAsColon → "202:43"
   249→        workedHrs: totalWorkHoursDecimal,
   250→        otHours,
   251→        otAmount,
   252→        grossSalary,
   253→        deductions: p.totalDeductions || 0,
   254→        bonus: p.bonus || 0,
   255→        netSalary,
   256→        status: p.status || 'generated',
   257→      };
   258→    }));
   259→
   260→    // ── Aggregate totals ──
   261→    const totalGross    = enriched.reduce((s, p) => s + p.grossSalary, 0);
   262→    const totalOTAmount = enriched.reduce((s, p) => s + p.otAmount, 0);
   263→    const totalBonus    = enriched.reduce((s, p) => s + p.bonus, 0);
   264→    const totalDeduct   = enriched.reduce((s, p) => s + p.deductions, 0);
   265→    const totalNet      = enriched.reduce((s, p) => s + p.netSalary, 0);
   266→    const totalOTHrs    = enriched.reduce((s, p) => s + p.otHours, 0);
   267→    const empProcessed  = enriched.length;
   268→    const avgNet        = empProcessed > 0 ? totalNet / empProcessed : 0;
   269→

   274→
   275→    const headerRows: any[][] = [
   276→      // Row 1
   277→      ['LAXREE GROUP OF COMPANIES', ...Array(14).fill('')],
   278→      // Row 2
   279→      [`Payroll Register — ${monthName} ${year}`, ...Array(14).fill('')],
   280→      // Row 3 — meta row: Generated date | (blank,blank) | Total Employees: X | (blank,blank) | Total Net Payroll: ₹Y
   281→      [`Generated: ${generatedDate}`, '', '', `Total Employees: ${empProcessed}`, '', '', `Total Net Payroll: ₹${totalNet.toLocaleString('en-IN', { maximumFractionDigits: 0 })}`, ...Array(8).fill('')],
   282→      // Row 4 — column letters (matches template)
   283→      [...cols],
   284→      // Row 5 — column headers

   288→        'Gross Salary', 'Deductions', 'Bonus', 'Net Salary', 'Status',
   289→      ],
   290→    ];
   291→
   292→    const ws1 = XLSX.utils.aoa_to_sheet(headerRows);
   293→
   294→    // ── Shared style definitions ──
   295→    const styleCompanyHeader = {
   296→      font: { bold: true, color: { rgb: GOLD }, sz: 18, name: 'Calibri' },
   297→      fill: { fgColor: { rgb: DARK } },
   298→      alignment: { horizontal: 'center' as const, vertical: 'center' as const },
   299→      border: {
   300→        top:    { style: 'medium' as const, color: { rgb: GOLD } },
   301→        bottom: { style: 'medium' as const, color: { rgb: GOLD } },
   302→        left:   { style: 'medium' as const, color: { rgb: GOLD } },
   303→        right:  { style: 'medium' as const, color: { rgb: GOLD } },
   304→      },
   305→    };
   306→
   307→    const styleReportTitle = {
   308→      font: { bold: true, color: { rgb: WHITE }, sz: 13, name: 'Calibri' },

   310→      alignment: { horizontal: 'center' as const, vertical: 'center' as const },
   311→      border: fullBorder('4472C4', 'thin'),

   313→
   314→    const styleMetaLabel = {

   327→
   328→    const styleColHeader = {

   492→

   495→    // ════════════════════════════════════════════════════════════
   496→    const sumRows: any[][] = [
