    18→function getFirmFromEmployeeId(employeeId: string): string {
    19→  const id = employeeId.toUpperCase();
    20→  if (id.startsWith('LAPL')) return 'LAPL';
    21→  if (id.startsWith('LRSL')) return 'LRSL';
    22→  if (id.startsWith('SI-') || id.startsWith('SI0')) return 'SI';
    23→  if (id.startsWith('SDF')) return 'SDF';
    24→  return '';
    25→}

    27→function formatHours(decimal: number): string {
    28→  if (!decimal || decimal === 0) return '0:00';
    29→  const hours = Math.floor(decimal);
    30→  const minutes = Math.round((decimal - hours) * 60);
    31→  if (minutes >= 60) return `${hours + 1}:00`;
    32→  return `${hours}:${String(minutes).padStart(2, '0')}`;
    33→}
    34→
    35→// Format overtime in clear human-readable format (e.g., "7m", "1h 30m")
    36→function formatOT(decimal: number): string {
    37→  if (!decimal || decimal === 0) return '0m';
    38→  const totalMinutes = Math.round(decimal * 60);
    39→  if (totalMinutes < 60) return `${totalMinutes}m`;
    40→  const hours = Math.floor(totalMinutes / 60);
    41→  const minutes = totalMinutes % 60;
    42→  if (minutes === 0) return `${hours}h`;
    43→  return `${hours}h ${minutes}m`;
    44→}
    45→
    46→// Format minutes directly to HH.MM string (e.g., 71 → "1.11", 325 → "5.25")
    47→function formatMinutesToHHMM(totalMinutes: number): string {
    48→  if (!totalMinutes || totalMinutes === 0) return '0.00';
    49→  const hours = Math.floor(totalMinutes / 60);
    50→  const minutes = totalMinutes % 60;
    51→  return `${hours}.${String(minutes).padStart(2, '0')}`;
    52→}
    53→
    54→// Convert HH.MM numeric (e.g., 1.11 = 1h 11min) to total minutes
    55→function hhmmToMinutes(val: number): number {
    56→  const h = Math.floor(val);
    57→  const m = Math.round((val - h) * 100); // extract minutes part from HH.MM
    58→  return h * 60 + m;
    59→}

    63→  return attendance.filter(a => a.isSunday && ['present', 'late', 'early-out', 'half-day', 'half_day'].includes(a.status)).length;

   175→      where: { employeeId, date: { gte: startDate, lt: endDate } },

   177→    });

   179→    const leaves = await db.leave.findMany({

   182→
   183→    const holidays = await db.holiday.findMany({ where: { date: { gte: startDate, lt: endDate } } });
   184→
   185→    // ── Effective cutoff day: caps future days and post-relieving days ──
   186→    const cutoffDay = getEffectiveCutoffDay(year, month, daysInMonth, employee.relievingDate);
   187→    const elapsedHolidays = countHolidaysUpTo(holidays, cutoffDay);
   188→    const holidayDays = elapsedHolidays;
   189→    // Sundays only up to cutoff day
   190→    const sundays = countSundaysUpTo(year, month, cutoffDay);

   194→    const rawPresentDays = effectiveAttendance.filter(a => ['present', 'late', 'early-out'].includes(a.status)).length;
   195→    const halfDays = effectiveAttendance.filter(a => a.status === 'half-day' || a.halfDay).length;

   203→        totalBaseHours += baseHrs;
   204→        if (a.status === 'half-day' || a.status === 'half_day') {
   205→          effectivePresentDays += 0.5;
   206→        } else {
   207→          effectivePresentDays += Math.min(1, baseHrs / employee.shiftHours);
   208→        }
   209→      }
   210→    }
   211→    effectivePresentDays = Math.round(effectivePresentDays * 100) / 100;
   212→    // Working days = cutoffDay - sundays - elapsedHolidays (caps future days)
   213→    const totalWorkingDays = Math.max(0, cutoffDay - sundays - elapsedHolidays);

   225→        presentDateStrs.add(`${ad.getFullYear()}-${String(ad.getMonth() + 1).padStart(2, '0')}-${String(ad.getDate()).padStart(2, '0')}`);

   234→      const end = new Date(leave.endDate);

   236→      const effectiveEnd = end > cutoffDate ? cutoffDate : end;
   237→      while (d <= effectiveEnd) {

   239→        const isSunday = d.getDay() === 0;
   240→        const isHoliday = holidayDateStrs.has(dateStr);

   242→          if (isUnpaid) effectiveUnpaidLeaves++;
   243→          else effectivePaidLeaves++;
   244→        }
