    48→// ── Convert decimal hours to HH:MM display format ──
    49→// e.g., 6.25 → "6:15", 9.5 → "9:30", 8.0 → "8:00"
    50→// Uses colon (:) separator so it's clearly hours:minutes, NOT a decimal fraction
    51→function formatHours(decimal: number): string {
    52→  if (!decimal || decimal === 0) return '0:00';
    53→  const hours = Math.floor(decimal);
    54→  const minutes = Math.round((decimal - hours) * 60);
    55→  if (minutes >= 60) return `${hours + 1}:00`;
    56→  return `${hours}:${String(minutes).padStart(2, '0')}`;
    57→}
    58→
    59→// ── Format overtime hours in clear human-readable format ──
    60→// Under 1 hour: shows minutes only (e.g., 7m, 20m)
    61→// 1+ hours: shows hours and minutes (e.g., 1h 30m, 2h 15m)
    62→// This avoids the confusing decimal display like "0.07h" which looks like 4.2 min but means 7 min
    63→function formatOT(decimal: number): string {
    64→  if (!decimal || decimal === 0) return '0m';
    65→  const totalMinutes = Math.round(decimal * 60);
    66→  if (totalMinutes < 60) {
    67→    return `${totalMinutes}m`;
    68→  }
    69→  const hours = Math.floor(totalMinutes / 60);
    70→  const minutes = totalMinutes % 60;
    71→  if (minutes === 0) return `${hours}h`;
    72→  return `${hours}h ${minutes}m`;
    73→}

    75→// ── Display a value that's already in HH:MM format ──
    76→function displayHHMM(value: number | undefined | null): string {
    77→  if (!value && value !== 0) return '0:00';
    78→  return formatHours(value);
    79→}

   170→// ── Animated Counter ──
   171→function AnimatedCounter({ value, prefix = '', suffix = '', decimals = 0 }: {
   172→  value: number; prefix?: string; suffix?: string; decimals?: number;
   173→}) {
   174→  const [display, setDisplay] = useState(0);
   175→  useEffect(() => {
   176→    const duration = 1000;
   177→    const startTime = Date.now();
   178→    const step = () => {
   179→      const elapsed = Date.now() - startTime;
   180→      const progress = Math.min(elapsed / duration, 1);
   181→      const eased = 1 - Math.pow(1 - progress, 3);
   182→      setDisplay(value * eased);
   183→      if (progress < 1) requestAnimationFrame(step);
   184→    };
   185→    requestAnimationFrame(step);
   186→  }, [value]);
   187→
   188→  if (decimals > 0) {
   189→    return <span>{prefix}{display.toFixed(decimals)}{suffix}</span>;
   190→  }
   191→  return <span>{prefix}{Math.round(display).toLocaleString()}{suffix}</span>;
   192→}

   497→  // ── Convert Excel serial date to YYYY-MM-DD ──
   498→  const excelSerialToDate = (serial: number): string => {
   499→    // Excel serial date: days since Jan 1, 1900 (with the 1900 leap year bug)
   500→    const epoch = new Date(1899, 11, 30); // Dec 30, 1899 is day 0
   501→    const date = new Date(epoch.getTime() + serial * 86400000);
   502→    const y = date.getFullYear();
   503→    const m = String(date.getMonth() + 1).padStart(2, '0');
   504→    const d = String(date.getDate()).padStart(2, '0');
   505→    return `${y}-${m}-${d}`;
   506→  };
   507→
   508→  // ── Convert Excel decimal time to HH:MM ──
   509→  const excelDecimalToTime = (value: any): string => {
   510→    if (typeof value === 'string') {
   511→      // Already a string - check if it looks like HH:MM
   512→      const trimmed = value.trim();
   513→      if (/^\d{1,2}:\d{2}(:\d{2})?$/.test(trimmed)) return trimmed.substring(0, 5);
   514→      // Try parsing as number
   515→      const num = parseFloat(trimmed);
   516→      if (!isNaN(num) && num >= 0 && num < 1) {
   517→        const totalMinutes = Math.round(num * 1440);
   518→        const hours = Math.floor(totalMinutes / 60);
   519→        const minutes = totalMinutes % 60;
   520→        return `${String(hours).padStart(2, '0')}:${String(minutes).padStart(2, '0')}`;
   521→      }
   522→      return trimmed;
   523→    }
   524→    if (typeof value === 'number') {
   525→      if (value >= 1) {
   526→        // Could be a full datetime serial - extract time portion
   527→        const timePart = value - Math.floor(value);
   528→        const totalMinutes = Math.round(timePart * 1440);
   529→        const hours = Math.floor(totalMinutes / 60);
   530→        const minutes = totalMinutes % 60;
   531→        return `${String(hours).padStart(2, '0')}:${String(minutes).padStart(2, '0')}`;
   532→      }
   533→      // Decimal time fraction (0 to <1)
   534→      const totalMinutes = Math.round(value * 1440);
   535→      const hours = Math.floor(totalMinutes / 60);
   536→      const minutes = totalMinutes % 60;
   537→      return `${String(hours).padStart(2, '0')}:${String(minutes).padStart(2, '0')}`;
   538→    }
   539→    return String(value || '');
   540→  };
   541→
   542→  // ── Parse any date value to YYYY-MM-DD ──
   543→  const parseDateValue = (value: any): string => {
   544→    if (!value && value !== 0) return '';
   545→    if (typeof value === 'number') {
   546→      // Excel serial date number
   547→      return excelSerialToDate(value);
   548→    }
   549→    const str = String(value).trim();
   550→    if (!str) return '';
   551→    // Already in YYYY-MM-DD format
   552→    if (/^\d{4}-\d{2}-\d{2}$/.test(str)) return str;
   553→    // Try DD/MM/YYYY or DD-MM-YYYY
   554→    const dmyMatch = str.match(/^(\d{1,2})[/\-.](\d{1,2})[/\-.](\d{4})$/);
   555→    if (dmyMatch) return `${dmyMatch[3]}-${dmyMatch[2].padStart(2, '0')}-${dmyMatch[1].padStart(2, '0')}`;
   556→    // Try MM/DD/YYYY
   557→    const mdyMatch = str.match(/^(\d{1,2})[/\-.](\d{1,2})[/\-.](\d{4})$/);
   558→    if (mdyMatch) return `${mdyMatch[3]}-${mdyMatch[1].padStart(2, '0')}-${mdyMatch[2].padStart(2, '0')}`;
   559→    // Fallback: try native Date parse
   560→    try {
   561→      const d = new Date(str);
   562→      if (!isNaN(d.getTime())) {
   563→        return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
   564→      }
   565→    } catch {}
   566→    return str;
   567→  };

   569→  // ── Smart Employee ID matching ──
   570→  const smartMatchEmployeeId = (rawId: string, empName: string): string => {
   571→    if (!rawId && !empName) return '';
   572→    let empId = String(rawId).trim();
   573→
   574→    // If we have an ID, try exact match first
   575→    if (empId) {
   576→      // Try exact match
   577→      const exact = employees.find(e => e.employeeId === empId);
   578→      if (exact) return exact.employeeId;
   579→
   580→      // Try with EMP- prefix (e.g., "41" → "EMP-041" or "EMP-41")
   581→      const withPrefix = `EMP-${empId.padStart(3, '0')}`;
   582→      const matchPad = employees.find(e => e.employeeId === withPrefix);
   583→      if (matchPad) return matchPad.employeeId;
   584→
   585→      const withoutPad = `EMP-${empId}`;
   586→      const matchNoPad = employees.find(e => e.employeeId === withoutPad);
   587→      if (matchNoPad) return matchNoPad.employeeId;
   588→    }
   589→
   590→    // Match by employee name
   591→    if (empName) {
   592→      const nameLower = empName.toLowerCase().trim();
   593→      // Exact name match
   594→      const exactName = employees.find(e => e.fullName.toLowerCase() === nameLower);
   595→      if (exactName) return exactName.employeeId;
   596→      // Partial name match
   597→      const partialName = employees.find(e =>
   598→        e.fullName.toLowerCase().includes(nameLower) || nameLower.includes(e.fullName.toLowerCase())
   599→      );
   600→      if (partialName) return partialName.employeeId;
   601→    }
   602→
   603→    return empId; // Return whatever we have
   604→  };
